import{readFileSync,writeFileSync,readdirSync}from'node:fs';import{spawnSync}from'node:child_process';
const r=JSON.parse(readFileSync('/tmp/mc02-rehearsal-location.json','utf8'));const name=process.argv[2];const pane='w31:p1D';const launch=JSON.parse(readFileSync(r.dir+'launch.json.txt','utf8'));if(Date.now()>Date.parse(launch.subjectDeadline))throw Error('Wall budget exhausted');const before=spawnSync('herdr',['agent','get',pane],{encoding:'utf8',timeout:10000});writeFileSync(r.dir+name+'-identity-before.json.txt',before.stdout);const identity=JSON.parse(before.stdout).result.agent;if(identity.name!=='mc02-subject-08'||identity.pane_id!==pane||!['idle','done'].includes(identity.agent_status))throw Error('Exact subject/idle gate failed');
const prompt=name==='compact'?'/compact':readFileSync(r.dir+name+'.prompt.txt','utf8');const start=new Date().toISOString();if(name==='compact')writeFileSync(r.dir+'compact.prompt.txt',prompt+'\n',{flag:'wx'});const p=spawnSync('herdr',['agent','prompt',pane,prompt,'--wait','--timeout','180000'],{encoding:'utf8',timeout:195000,maxBuffer:2*1024*1024});writeFileSync(r.dir+name+'-operation.json.txt',JSON.stringify({startedAt:start,endedAt:new Date().toISOString(),exit:p.status,error:p.error?.message??null,stdout:p.stdout,stderr:p.stderr},null,2)+'\n',{flag:'wx'});console.log(p.stdout,p.stderr);

if(calls>6||cmps.length>1||(promptTokens!==null&&promptTokens>32000)||Date.now()>Date.parse(launch.subjectDeadline))throw Error('Budget exceeded: STOP');if(messages.some(e=>e.message.provider!=='zai'||e.message.model!=='glm-5.3-flash'))throw Error('Model changed: STOP');if(tools.some(t=>t.name!=='read'||t.arguments?.path!==r.cwd+'/checkpoint.md'))throw Error('Unexpected tool attempt: STOP');if(p.status!==0)process.exitCode=1;
